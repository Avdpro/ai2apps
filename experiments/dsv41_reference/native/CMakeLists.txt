cmake_minimum_required(VERSION 3.27)
project(dsv41_native_loader LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -m nanobind --cmake_dir OUTPUT_VARIABLE nanobind_ROOT OUTPUT_STRIP_TRAILING_WHITESPACE)
find_package(nanobind CONFIG REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -m mlx --cmake-dir OUTPUT_VARIABLE MLX_ROOT OUTPUT_STRIP_TRAILING_WHITESPACE)
find_package(MLX CONFIG REQUIRED)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(GLM_SOURCE "${REPO_ROOT}/omlx/custom_kernels/glm_moe_dsa/csrc")
nanobind_add_module(_dsv41_loader NB_STATIC NB_DOMAIN mlx bindings.cpp "${GLM_SOURCE}/expert_loader.cpp")
target_include_directories(_dsv41_loader PRIVATE "${GLM_SOURCE}")
target_link_libraries(_dsv41_loader PRIVATE mlx)
